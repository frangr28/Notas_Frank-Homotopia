\section*{Preliminares (de Topología)}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.]
    Un espacio \(X\) es \textit{Hausdorff\index{espacio Hausdorff, \(T_2\)}} o \(T_2\) si 
    para todo \(x_1, x_2\in X\) tales que \(x_1\neq x_2\), existen abiertos \(\mathcal{U}_1, \mathcal{U}_2\) 
    tal que \(x_i\in\mathcal{U}_i\) y \(\mathcal{U}_1\cap\mathcal{U}_2 =\varnothing\).
\end{tcolorbox}

\begin{tcolorbox}[colback=antiquefuchsia!15!white,colframe=antiquefuchsia!75!black,title=\textbf{Convención}]
    Todos nuestros espacios son Hausdorff.
\end{tcolorbox}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.]
    Dado un espacio \(X\), una \textit{cubierta abierta (cerrada)\index{cubierta abierta}} es una familia de conjuntos abiertos (cerrados) 
    \(\{U_\lambda\}_{\lambda\in\Lambda}\) tal que \(X=\displaystyle\bigcup_{\lambda\in\Lambda}\mathcal{U}_\lambda\).
\end{tcolorbox}

\begin{tcolorbox}[colback=antiquefuchsia!15!white,colframe=antiquefuchsia!75!black,title=\textbf{Observación}]
    Las siguientes afirmaciones son equivalentes:
    \begin{enumerate}[noitemsep, topsep=0pt]
        \item Un espacio es \textit{compacto}\index{espacio compacto} si para toda cubierta existe una subcubierta.
        \item Toda familia de cerrados cuya intersección es vacía, contiene una subfamilia cuya intersección es vacía.
        \item Toda familia de cerrados tal que toda subfamilia finita tiene intersección no vacía, tiene ella misma intersección no vacía. 
    \end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!blue,title=\textbf{Teorema}.]
    \begin{itemize}[noitemsep, topsep=0pt]
        \item Si \(A\subseteq X\), con \(A\) compacto y \(X\) es \(T_2\), entonces \(A\) es cerrado.
        \item Si \(X\) es \(T_2\) compacto, entonces \(A\subseteq X\) es compacto si y sólo si \(A\) es cerrado.
    \end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.]
    Un espacio \(X\) es \textit{compactamente generado\index{espacio compactamente generado}} si se cumple la siguiente propiedad: 
    
    % \begin{center}
    \(C\subseteq X\) es cerrado syss \(C\cap K\) es cerrado, para todo \(K\subseteq X\) compacto.
    % \end{center}
\end{tcolorbox}

% \begin{tcolorbox}
%     Si no se pide esta propiedad en \(\mathsf{Top}\), el producto puede que no nos quede \(\textsf{CW}\).
% \end{tcolorbox}

\begin{tcolorbox}
    Queremos que todos nuestros espacios sean compactamente generados.
    \tcblower
    Dado \(X\) un espacio \(T_2\), 
    \[\mathscr{C}=\{A\subseteq X \mid A\cap K \text{ es cerrado en } X, \text{ para todo } K\subseteq X \text{ compacto}\}\]
    es un conjunto de cerrados tales que la topología generada es compactamente generada, y cualquier función continua \(f\colon X\to Y\) sigue siendo continua.
\end{tcolorbox}

\begin{tcolorbox}[colback=antiquefuchsia!15!white,colframe=antiquefuchsia!75!black,title=\textbf{Convención}]
    Desde ahora, todos nuestros espacios son compactamente generados.
\end{tcolorbox}

\begin{tcolorbox}[colback=antiquefuchsia!15!white,colframe=antiquefuchsia!75!black,title=\textbf{Convención}]
    Diremos \textit{función continua} o \textit{mapa}. \index{mapa}\index{función continua} 
    \begin{itemize}[noitemsep, topsep=0pt]
        \item[] \(\operatorname{Hom}(X, Y)= \operatorname{Map}(X, Y)=M(X, Y)= \{f\colon X\to Y \mid f \text{ función continua}\}\).

        \item[] \(\operatorname{Hom_{\ast}}((X, x_0), (Y, y_0))= \operatorname{Map_{\ast}}((X, x_0), (Y, y_0))=M_{\ast}((X, x_0), (Y, y_0))\)
    
        % = \{f\colon (X, x_0) \to (Y, y_0) \mid f \text{ función continua}\}\).
    \end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.]
    \begin{enumerate}[noitemsep, topsep=0pt]
        \item[] \(\mathsf{Top}\) es la categoría de espacios topológicos. \index{categoría de espacios topológicos}
        \item[] \(\mathsf{Top}_{\ast}\) es la categoría de espacios topológicos basados.
    \end{enumerate}
    \tcblower 
    Los morfismos son los mapas o funciones continuas (basadas).
    
    Esto es, \((X, A) \rightarrow (Y, B)\) con \(f(A)\subseteq B\).
    
    En el caso basado, \((X, x_0)\rightarrow (Y, y_0)\) dado por \(f(x_0)=y_0\).
\end{tcolorbox}


\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.]
    Sea \(X\) un conjunto y \(\{W_{\alpha}\}_{\alpha\in A}\) una familia de espacios tales que \(W_{\alpha}\subseteq X\).

    La \textit{topología débil}\index{topología débil} de \(X\), está dada por,
    
    \begin{center}
        \(\mathcal{U}\subseteq X\) es un abierto si y sólo si \(\mathcal{U}\cap W_{\alpha}\) es abierto, para todo \(\alpha\in A\).
    \end{center}
\end{tcolorbox}

\begin{tcolorbox}
    \[X_{+}=X\sqcup\{\text{puntos que no están en } X\}\]
\end{tcolorbox}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.]
    Sea \(X\) un espacio, \(Q\) un conjunto y \(q\colon X\to Q\).
    
    La \textit{topología cociente} (o de \textit{identificación}), \index{topología cociente, de identificación} está dada por,
    
    \begin{center}
        \(\mathcal{U}\subseteq Q\) es abierto (cerrado) si y sólo si \(q^{-1}(\mathcal{U})\) es abierto (cerrado).
    \end{center}
\end{tcolorbox}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.]
    Sean \(X, Y\) espacios, \(p\colon X\to Y\). El mapa es \textit{cociente} o \textit{de identificación}\index{mapa cociente, de identificación}
    si la topología de \(Y\) coincide con la topología que genera como conjunto, denotada \(\mathcal{T}(p)\).
\end{tcolorbox}

\section*{Homotopía}
\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.\index{mapas homotópicos, funciones homotópicas}]
    Si \(f, g\colon X\to Y\) son mapas, decimos que son \textit{homotópicas}, en símbolos \(f\simeq g\), 
    
    si existe \(H\colon X\times I \to Y\) tal que 
    \(\qquad 
    \begin{aligned}
        H(x, 0)&\sim f(x)\\[-3pt]
        H(x, 1)&\sim g(x)\\
    \end{aligned}\)
\tcblower
    Si los espacios son basados, para toda \(t\in I\) se tiene que \(H(\ast, t)=\ast_{Y}\)
\end{tcolorbox}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!blue,title=\textbf{Ley exponencial}.\index{ley exponencial}]
    \(\operatorname{Map}(X\times I, Y)\cong\operatorname{Map}(I, \operatorname{Map}(X, Y))\)
    \tcblower
    \(Y^{X\times I}\cong(Y^X)^I\)
\end{tcolorbox}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!blue,title=\textbf{Proposición}.]
    Homotopía es una relación de equivalencia.
\tcblower
    Además, si \(f\simeq g\colon X\to Y, k\colon A\to X, \ell\colon Y\to B\), entonces \(f\circ k\simeq g\circ k\) y \(\ell\circ f \simeq \ell \circ g\).
\end{tcolorbox}

\begin{tcolorbox}\index{concatenación}
    Si \(F, G\colon X\times I \to Y\) homotopías tales que \(F(x, 1)=G(x, 0)\), se tiene la \textit{concatenación} \(H=F\ast G\).
\end{tcolorbox}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.\index{clase de equivalencia}]
    Para \(f\colon X\to Y\) continua, \([f]\) es la \textit{clase de equivalencia de \(f\)}, es decir, el conjunto de todas las funciones continuas de \(X\to Y\) que son homotópicas a \(f\).
\end{tcolorbox}

\begin{tcolorbox}[colback=antiquefuchsia!15!white,colframe=antiquefuchsia!75!black,title=Notación]    
    \([X, Y], [X, Y]_{\ast}\), la colección de todas las clases de equivalencia que nos dan las homotopías para nuestros mapas de \(X\) a \(Y\).
\end{tcolorbox}

\begin{tcolorbox}[colback=antiquefuchsia!15!white,colframe=antiquefuchsia!75!black,title=\textbf{Observación}\index{precomposición}]
    \begin{enumerate}[noitemsep, topsep=0pt]
        \item Si \(\alpha=[f]\in[X, Y]_{\_}\) y \(\beta=[g]\in[Y, Z]\), entonces \(\beta\alpha=[g\circ f]\in [X, Z]\).
        \item Si \(X, Y\) son basados, \(C_\ast\colon X\to Y\), entonces \(0=[C_\ast]\in[X, Y]_{\ast}\).
        \item \(k\colon A\to X, \ell\colon Y\to B\), se tiene la \textit{precomposición}
            \begin{itemize}[noitemsep, topsep=0pt]
                \item[] \(k^{\ast}\colon \operatorname{Map_\_}(X, Y)\to \operatorname{Map}(A, Y)\) dada por \(f\mapsto f\circ k = k^{\ast}(f)\).
                \item[] \(\ell_{\ast}\colon \operatorname{Map_\_}(X, Y)\to \operatorname{Map_\_}(Y, B)\) dada por \(f\mapsto \ell\circ f = \ell^{\ast}(f)\).
            \end{itemize}
            \begin{itemize}[noitemsep, topsep=0pt]
                \item[] \(k^{\ast}\colon [X, Y]_\_\to [A, Y]_\_\)
                \item[] \(\ell_{\ast}\colon [X, Y]_\_\to [X, B]_\_\)
            \end{itemize}
    \end{enumerate}
    \tcblower
    \(k^{\ast}(0)=0, \ell_{\ast}(0)=0\)
\end{tcolorbox}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!blue,title=\textbf{Hecho}.]
    Si \(k\simeq k', \ell\simeq\ell'\), entonces en las clases de equivalencia, \(k^{\ast}\simeq k'^{\ast}, \ell^{\ast}\simeq\ell'^{\ast}\).
\end{tcolorbox}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!blue,title=\textbf{Ley exponencial para el smash}.\index{ley exponencial para el smash}]
    \(\operatorname{Map_\ast}(X\wedge Y, Z\cong\operatorname{Map_\ast}(X, \operatorname{Map_\ast}(Y, Z))\)

    \(\operatorname{Map_\ast}(X\wedge I_{+})\cong\operatorname{Map_\ast}(I_{+}, \operatorname{Map_\ast}(X, Y))\)
\end{tcolorbox}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!blue,title=\textbf{Proposición}.]
    Si \(f, g\colon X\to Y\) mapas. Entonces

    \(f\simeq g\) si y sólo si existe \(G\colon X\to Y^I\) tal que     
    \(\qquad 
    \begin{aligned}
        G(x)(0)&=f(x)\\[-3pt]
        G(x)(1)&=g(x)\\[-3pt]
        G(\ast)(1)&=\ast\\
    \end{aligned}\)
\end{tcolorbox}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.\index{tipo de homotopía}]
    Decimos que \(X\) y \(Y\) \textit{tienen el mismo tipo de homotopía}, en símbolos \(X\simeq Y\),
    
    si existen \(f\colon X\to Y\), \(g\colon Y\to X\) tales que \(g\circ f\simeq 1_{X}\) y \(f\circ g\simeq 1_{Y}\).
\tcblower
    En este caso, a \(f\) y \(g\) se les llama \textit{equivalencias homotópicas},\index{equivalencias homotópicas}, también decimos que 
    
    \(X\) es \textit{homotópicamente equivalente} a \(Y\).\index{homotópicamente equivalente}
\end{tcolorbox}


\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.\index{nulo homotopía}]
    \(f\colon X\to Y\) es \textit{nulo homotópica} si \(f\simeq\ast\colon X\to Y\).
\tcblower
\(\ast\) función constante
\end{tcolorbox}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.\index{espacio contraíble, espacio contráctil}]
    Un espacio \(X\) es \textit{contraíble} o \textit{contráctil} si \(X\simeq \ast\).
\end{tcolorbox}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!blue,title=\textbf{Lema}.]
    \begin{enumerate}[noitemsep, topsep=0pt]
        \item Sean \(A, X, Y\), \(f_0, f_1\in[f_1]\in[A, X]\) y \(g_1\simeq g_0\colon A\to Y\), 
        
        entonces \((f_0, g_0)\simeq (f_1, g_1)\colon A\to X\times Y\).
        \item Sean \(B, X, Y\) espacios, \(f_0\simeq f_1\colon X\to B\), \(g_1\simeq g_0\colon Y\to B\), 
        
        entonces \(f_0\vee g_0\simeq f_1\vee g_1\colon X\vee Y\to B\).
    \end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!blue,title=\textbf{Corolario}.]
    \begin{enumerate}[noitemsep, topsep=0pt]
        \item \([A, X]_\_\times [A, Y]\to[A, X\times Y]_\_\) dada por \(([f], [g])\mapsto [(f, g)]\) es una biyección.
        \item \([X, B]_\ast\times [Y, B]_\ast\to[X\vee Y, B]_\ast\) dada por \(([f], [g])\mapsto [f \vee g]\) es una biyección.
    \end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.\index{retracción}]

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.\index{sección}]
    Sea \(f\colon X\to Y\), una \textit{sección} de \(f\) es un mapa \(s\colon Y\to X\) tal que \(f\circ s\simeq 1_{Y}\).
\tcblower
    \begin{center}  
    \(\begin{tikzcd}

        & Y \arrow[l, bend right, swap]{}{s}

    \end{center}
\end{tcolorbox}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.\index{retracción homotópica}\index{inversa homotópica izquierda}]
    Una \textit{retracción homotópica}, o una \textit{inversa homotópica izquierda} de \(f\colon X\to Y\)
    es un mapa \(g\colon Y\to X\) tal que \(g\circ f\simeq 1_{X}\). 
    \tcblower
    Decimos que \(X\) está \textit{dominado} por \(Y\).
\end{tcolorbox}

\begin{tcolorbox}[colback=blue-violet!5!white,colframe=blue-violet!75!black,title=\textbf{Definición}.\index{sección homotópica}\index{inversa homotópica derecha}]
    Una \textit{sección homotópica}, o una \textit{inversa homotópica derecha} de \(f\colon X\to Y\)   
    es un mapa \(h\colon Y\to X\) tal que \(f\circ h\simeq 1_{Y}\). 
\end{tcolorbox}
